   168→function slugify(input: string): string {
   169→  return input
   170→    .toLowerCase()
   171→    .normalize("NFD")
   172→    .replace(/[\u0300-\u036f]/g, "")
   173→    .replace(/[^a-z0-9]+/g, "-")
   174→    .replace(/(^-|-$)+/g, "");
   175→}

  1074→function formFromProduct(p: Product): ProductFormState {
  1075→  const ss = p.sizeStock ?? {};
  1076→  const sizeStock: Record<string, string> = {};
  1077→  for (const s of p.sizes) {
  1078→    const v = ss[String(s)];
  1079→    sizeStock[String(s)] = v !== undefined ? String(v) : "";
  1080→  }
  1081→  return {
  1082→    name: p.name,
  1083→    slug: p.slug,
  1084→    brand: p.brand,
  1085→    category: p.category,
  1086→    price: String(p.price),
  1087→    stock: String(p.stock),
  1088→    rating: String(p.rating),
  1089→    accent: p.accent,
  1090→    badge: p.badge ?? "",
  1091→    description: p.description,
  1092→    sizes: [...p.sizes].sort((a, b) => a - b),
  1093→    sizeStock,
  1094→    images: [...p.images],
  1095→    featured: p.featured,
  1096→    bestSeller: p.bestSeller,
  1097→  };

  1163→  function handleFiles(files: File[]) {
  1164→    const imageFiles = files.filter((f) => f.type.startsWith("image/"));
  1165→    if (imageFiles.length === 0) {
  1166→      toast.error("Selecione apenas arquivos de imagem.");

  1173→    }
  1174→    const toRead = imageFiles.slice(0, availableSlots);
