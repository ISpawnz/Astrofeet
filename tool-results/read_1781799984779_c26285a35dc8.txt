   141→function slugify(input: string): string {
   142→  return input
   143→    .toLowerCase()
   144→    .normalize("NFD")
   145→    .replace(/[\u0300-\u036f]/g, "")
   146→    .replace(/[^a-z0-9]+/g, "-")
   147→    .replace(/(^-|-$)+/g, "");
   148→}

   557→
   558→function OrdersTable() {
   559→  const queryClient = useQueryClient();
   560→  const [query, setQuery] = useState("");
   561→  const [updating, setUpdating] = useState<string | null>(null);
   562→
   563→  const { data, isLoading, isError } = useQuery({
   564→    queryKey: ["orders"],
   565→    queryFn: () => api.listOrders(),
   566→  });
   567→
   568→  const filtered = useMemo(() => {
   569→    if (!data) return [];
   570→    const q = query.trim().toLowerCase();
   571→    if (!q) return data;
   572→    return data.filter(
   573→      (o) =>
   574→        o.code.toLowerCase().includes(q) ||
   575→        o.customer.name.toLowerCase().includes(q),
   576→    );
   577→  }, [data, query]);
   578→
   579→  async function onStatusChange(order: Order, status: string) {
   580→    setUpdating(order.id);
   581→    try {
   582→      await api.updateOrderStatus(order.id, status);
   583→      await Promise.all([
   584→        queryClient.invalidateQueries({ queryKey: ["orders"] }),
   585→        queryClient.invalidateQueries({ queryKey: ["admin-metrics"] }),
   586→      ]);
   587→      toast.success(`Pedido ${order.code} atualizado para "${orderStatusLabel(status)}".`);
   588→    } catch (err) {
   589→      toast.error(
   590→        err instanceof Error ? err.message : "Não foi possível atualizar o status.",
   591→      );
   592→    } finally {
   593→      setUpdating(null);
   594→    }
   595→  }
   596→
   597→  if (isError) {
   598→    return (
   599→      <div className="glass rounded-2xl border border-white/10 p-8 text-center text-sm text-muted-foreground">
   600→        Não foi possível carregar os pedidos. Tente novamente em instantes.
   601→      </div>
   602→    );
   603→  }
   604→
   605→  return (
   606→    <div className="glass rounded-2xl border border-white/10 p-4 sm:p-5">
   607→      <div className="mb-4 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   608→        <div>
   609→          <h3 className="font-semibold">Todos os pedidos</h3>
   610→          <p className="text-xs text-muted-foreground">
   611→            Atualize o status de cada pedido em tempo real.
   612→          </p>
   613→        </div>

   798→  const [saving, setSaving] = useState(false);

   807→
   808→  function update<K extends keyof ProductFormState>(
   809→    key: K,
   810→    value: ProductFormState[K],

   812→    setForm((f) => ({ ...f, [key]: value }));

   814→
   815→  function updateName(v: string) {
   816→    setForm((f) => {

   818→      if (!slugEdited) next.slug = slugify(v);

   823→  function toggleSize(s: number) {

   827→        ? f.sizes.filter((x) => x !== s)

   831→  }

   852→    const slug = (form.slug.trim() || slugify(name)).toLowerCase();
   853→    const imagesRaw = form.images
   854→      .split(",")
   855→      .map((s) => s.trim())
