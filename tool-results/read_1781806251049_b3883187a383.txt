   106→function paymentMethodInfo(method: string): {
   107→  label: string;
   108→  icon: React.ReactNode;
   109→} {
   110→  switch (method) {
   111→    case "card":
   112→      return { label: "Cartão de crédito", icon: <CreditCard className="h-4 w-4" /> };
   113→    case "pix":
   114→      return { label: "Pix", icon: <QrCode className="h-4 w-4" /> };
   115→    case "boleto":
   116→      return { label: "Boleto", icon: <Barcode className="h-4 w-4" /> };
   117→    default:
   118→      return { label: method || "Pagamento", icon: <CreditCard className="h-4 w-4" /> };
   119→  }
   120→}

   122→function getInitial(name: string | undefined): string {
   123→  if (!name) return "✦";
   124→  const trimmed = name.trim();
   125→  if (!trimmed) return "✦";
   126→  return trimmed[0]!.toUpperCase();
   127→}
   128→
   129→function itemsSummary(order: Order): string {
   130→  const items = order.items;
   131→  if (items.length === 0) return "Sem itens";
   132→  const firstTwo = items.slice(0, 2).map((i) => `${i.name} (Tam ${i.size})`);
   133→  const more = items.length - 2;
   134→  const base = firstTwo.join(", ");
   135→  if (more > 0) return `${base} +${more} mais`;
   136→  return base;
   137→}
   138→
   139→function itemCount(order: Order): number {
   140→  return order.items.reduce((n, i) => n + i.quantity, 0);
   141→}
   142→
   143→function isCancelled(status: string): boolean {
   144→  return status === "cancelled";

   775→function ProfileTab({ user }: { user: PublicUser }) {

   777→  const setUser = useAuthStore((s) => s.setUser);
   778→  const roleLabel = user.role === "admin" ? "Comando" : "Explorador";
   779→  const roleAccent =
   780→    user.role === "admin" ? "var(--neon-magenta)" : "var(--neon-cyan)";
   781→
   782→  // Editable name state
   783→  const [editingName, setEditingName] = useState(false);
   784→  const [nameValue, setNameValue] = useState(user.name);
   785→  const [savingName, setSavingName] = useState(false);
   786→
   787→  // Password change state
   788→  const [currentPassword, setCurrentPassword] = useState("");
   789→  const [newPassword, setNewPassword] = useState("");
   790→  const [confirmPassword, setConfirmPassword] = useState("");
   791→  const [savingPassword, setSavingPassword] = useState(false);
   792→  const [showCurrent, setShowCurrent] = useState(false);
   793→  const [showNew, setShowNew] = useState(false);
   794→  const [showConfirm, setShowConfirm] = useState(false);
   795→
   796→  useEffect(() => {
   797→    setNameValue(user.name);
   798→  }, [user.name]);
   799→
   800→  async function saveName() {
   801→    const trimmed = nameValue.trim();
   802→    if (trimmed.length < 2) {
   803→      toast.error("Nome precisa ter ao menos 2 caracteres.");
   804→      return;
   805→    }
   806→    if (trimmed === user.name) {
   807→      setEditingName(false);
   808→      return;
   809→    }
   810→    setSavingName(true);
   811→    try {
   812→      const updated = await api.updateProfile(trimmed);
   813→      setUser(updated);
   814→      toast.success("Nome atualizado com sucesso!");
   815→      setEditingName(false);
   816→    } catch (err) {
   817→      toast.error(
   818→        err instanceof Error ? err.message : "Não foi possível salvar.",
   819→      );
   820→    } finally {
   821→      setSavingName(false);
   822→    }
   823→  }

   825→  function resetPasswordFields() {

   827→    setNewPassword("");
   828→    setConfirmPassword("");

   833→      toast.error("Informe sua senha atual.");

   837→      toast.error("A nova senha precisa ter ao menos 6 caracteres.");
   838→      return;
   839→    }
   840→    if (newPassword === currentPassword) {
   841→      toast.error("A nova senha precisa ser diferente da atual.");
   842→      return;
   843→    }
   844→    if (newPassword !== confirmPassword) {
   845→      toast.error("As senhas não coincidem.");
   846→      return;
   847→    }
   848→    setSavingPassword(true);
   849→    try {
   850→      await api.changePassword(currentPassword, newPassword);

   852→      resetPasswordFields();
   853→    } catch (err) {
   854→      toast.error(
   855→        err instanceof Error
   856→          ? err.message
   857→          : "Não foi possível atualizar a senha agora.",
   858→      );
   859→    } finally {
   860→      setSavingPassword(false);
   861→    }
   862→  }
   863→
   864→  return (
