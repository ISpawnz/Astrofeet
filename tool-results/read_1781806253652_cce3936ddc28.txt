   234→   234→function Info({
   235→   235→  product,
   236→   236→  reviews,
   237→   237→}: {
   238→   238→  product: Product;
   239→   239→  reviews: Review[];
   240→   240→}) {
   241→   241→  const add = useCartStore((s) => s.add);
   242→   242→  const openCart = useCartStore((s) => s.open);
   243→   243→  const navigate = useUIStore((s) => s.navigate);
   244→   244→  const openSizeGuide = useUIStore((s) => s.openSizeGuide);
   245→   245→  const toggleWishlist = useWishlistStore((s) => s.toggleProduct);
   246→   246→  const inWishlist = useWishlistStore((s) => s.has(product.id));
   247→   247→  const [heartBump, setHeartBump] = useState(false);
   248→   248→
   249→   249→  const [size, setSize] = useState<number | null>(null);
   250→   250→  const [qty, setQty] = useState(1);
   251→   251→
   252→   252→  // Per-size-aware max quantity (fall back to global stock when sizeStock absent).
   253→   253→  const sizeStockMap = product.sizeStock ?? {};
   254→   254→  const selectedSizeStock =
   255→   255→    size !== null ? sizeStockMap[String(size)] : undefined;
   256→   256→  const effectiveMax =
   257→   257→    selectedSizeStock !== undefined ? selectedSizeStock : product.stock;
   258→   258→  const maxQty = Math.max(1, Math.min(effectiveMax, 99));
   259→   259→  const installment = product.price / 10;
   260→   260→  const freeShipping = product.price >= FREE_SHIPPING_THRESHOLD;
   261→   261→  const reviewCount = reviews.length;
   262→   262→
   263→   263→  // Reset qty if it exceeds the newly-selected size's stock.
   264→   264→  useEffect(() => {
   265→   265→    // eslint-disable-next-line react-hooks/set-state-in-effect
   266→   266→    if (qty > maxQty) setQty(Math.max(1, maxQty));
   267→   267→  }, [maxQty, qty]);
   268→   268→
   269→   269→  function selectSize(s: number) {
   270→   270→    setSize(s);
   271→   271→    setQty(1);
   272→   272→  }
   273→   273→
   274→   274→  function requireSize(): boolean {
   275→   275→    if (size === null) {
   276→   276→      toast.error("Selecione um tamanho para continuar.");
   277→   277→      return false;
   278→   278→    }
   279→   279→    return true;
   280→   280→  }
   281→   281→
   282→   282→  function handleAdd() {
   283→   283→    if (!requireSize()) return;
   284→   284→    add(product, size as number, qty);
   285→   285→    toast.success(`${product.name} adicionado ao carrinho`);
   286→   286→    openCart();
   287→   287→  }
   288→   288→
   289→   289→  function handleBuyNow() {
   290→   290→    if (!requireSize()) return;
   291→   291→    add(product, size as number, qty);
   292→   292→    navigate("checkout");
   293→   293→  }
   294→   294→
   295→   295→  function handleWishlist() {
   296→   296→    toggleWishlist(product);
   297→   297→    setHeartBump(true);
   298→   298→    setTimeout(() => setHeartBump(false), 450);
   299→   299→    toast.success(
   300→   300→      inWishlist
   301→   301→        ? `${product.name} saiu da sua lista`
   302→   302→        : `${product.name} salvo na lista de desejos`,
   303→   303→    );
   304→   304→  }
   305→   305→
   306→   306→  return (
   307→   307→    <motion.div
   308→   308→      initial={{ opacity: 0, y: 20 }}
   309→   309→      animate={{ opacity: 1, y: 0 }}
   310→   310→      transition={{ duration: 0.5, delay: 0.1 }}
   311→   311→      className="flex flex-col gap-5"
   312→   312→    >
   313→   313→      {/* Brand + wishlist */}
   314→   314→      <div className="flex items-start justify-between gap-3">
   315→   315→        <p className="text-xs font-semibold uppercase tracking-[0.2em] text-muted-foreground">
   316→   316→          {product.brand}
   317→   317→        </p>
   318→   318→        <button
   319→   319→          onClick={handleWishlist}
   320→   320→          className={cn(
   321→   321→            "flex h-10 w-10 shrink-0 items-center justify-center rounded-full border transition-all hover:scale-110",
   322→   322→            inWishlist
   323→   323→              ? "border-[var(--neon-magenta)]/50 bg-[var(--neon-magenta)]/20 text-[var(--neon-magenta)]"
   324→   324→              : "border-white/10 bg-white/5 text-foreground/70 hover:text-foreground",
   325→   325→          )}
   326→   326→          aria-label={
   327→   327→            inWishlist ? "Remover da lista de desejos" : "Salvar na lista de desejos"
   328→   328→          }
   329→   329→          aria-pressed={inWishlist}
   330→   330→        >
   331→   331→          <Heart
   332→   332→            className={cn("h-5 w-5", heartBump && "animate-heartbeat", inWishlist && "fill-current")}
   333→   333→          />
   334→   334→        </button>
   335→   335→      </div>
   336→   336→
   337→   337→      {/* Name */}
   338→   338→      <h1 className="text-3xl font-black leading-tight sm:text-4xl">
   339→   339→        {product.name}
   340→   340→      </h1>
   341→   341→
   342→   342→      {/* Rating */}
   343→   343→      <div className="flex items-center gap-2 text-sm">
   344→   344→        <Stars value={product.rating} size={16} />
   345→   345→        <span className="font-semibold">{product.rating.toFixed(1)}</span>
   346→   346→        <span className="text-muted-foreground">
   347→   347→          · {reviewCount}{" "}
   348→   348→          {reviewCount === 1 ? "avaliação" : "avaliações"}
   349→   349→        </span>
   350→   350→      </div>
   351→   351→
   352→   352→      {/* Price */}
   353→   353→      <div className="space-y-1">
   354→   354→        <p className="text-3xl font-bold">{formatPrice(product.price)}</p>
   355→   355→        <p className="text-sm text-muted-foreground">
   356→   356→          ou 10x de{" "}
   357→   357→          <span className="font-semibold text-foreground">
   358→   358→            {formatPrice(installment)}
   359→   359→          </span>{" "}
   360→   360→          sem juros
   361→   361→        </p>
   362→   362→        <p
   363→   363→          className={cn(
   364→   364→            "text-xs font-medium",
   365→   365→            freeShipping ? "text-[var(--neon-lime)]" : "text-muted-foreground",
   366→   366→          )}
   367→   367→        >
   368→   368→          {freeShipping
   369→   369→            ? "🚀 Frete grátis acima de R$300 — liberado!"
   370→   370→            : "Frete grátis em pedidos acima de R$300"}
   371→   371→        </p>
   372→   372→      </div>
   373→   373→
   374→   374→      <Separator className="bg-white/10" />
   375→   375→
   376→   376→      {/* Short description */}
   377→   377→      <p className="text-sm leading-relaxed text-muted-foreground">
   378→   378→        {product.description.split("\n")[0]?.slice(0, 220) ??
   379→   379→          "Modelo exclusivo Astrofeet, feito para quem anda entre estrelas."}
   380→   380→      </p>
   381→   381→
   382→   382→      {/* Size selector */}
   383→   383→      <div className="space-y-2">
   384→   384→        <div className="flex items-center justify-between">
   385→   385→          <label className="text-sm font-semibold">Tamanho</label>
   386→   386→          <button
   387→   387→            onClick={openSizeGuide}
   388→   388→            className="inline-flex items-center gap-1.5 text-xs font-medium text-[var(--neon-cyan)] transition hover:underline"
   389→   389→          >
   390→   390→            <Ruler className="h-3.5 w-3.5" />
   391→   391→            Guia de medidas
   392→   392→          </button>
   393→   393→        </div>
   394→   394→        <div className="flex flex-wrap gap-2">
   395→   395→          {product.sizes.map((s) => {
   396→   396→            const isSelected = size === s;
   397→   397→            // Per-size stock: prefer sizeStock map, fall back to global stock.
   398→   398→            const sizeStockMap = product.sizeStock ?? {};
   399→   399→            const perSize = sizeStockMap[String(s)];
   400→   400→            const effectiveStock =
   401→   401→              perSize !== undefined ? perSize : product.stock;
   402→   402→            const soldOut = effectiveStock <= 0;
   403→   403→            const lowStock = !soldOut && effectiveStock <= 2;
   404→   404→            return (
   405→   405→              <button
   406→   406→                key={s}
   407→   407→                onClick={() => !soldOut && selectSize(s)}
   408→   408→                disabled={soldOut}
   409→   409→                aria-pressed={isSelected}
   410→   410→                title={
   411→   411→                  soldOut
   412→   412→                    ? `Tamanho ${s} esgotado`
   413→   413→                    : lowStock
   414→   414→                      ? `Últimas ${effectiveStock} unidades`
   415→   415→                      : `Tamanho ${s}`
   416→   416→                }
   417→   417→                className={cn(
   418→   418→                  "relative h-12 min-w-14 rounded-xl border px-3 text-sm font-bold transition",
   419→   419→                  soldOut && "cursor-not-allowed opacity-40 line-through",
   420→   420→                  isSelected && !soldOut
   421→   421→                    ? "border-transparent text-black"
   422→   422→                    : lowStock
   423→   423→                      ? "border-amber-500/40 bg-amber-500/5 text-amber-200 hover:border-amber-500/60"
   424→   424→                      : "border-white/10 bg-white/[0.03] hover:border-white/30 hover:bg-white/[0.06]",
   425→   425→                )}
   426→   426→                style={
   427→   427→                  isSelected && !soldOut
   428→   428→                    ? {
   429→   429→                        background: product.accent,
   430→   430→                        boxShadow: `0 0 18px ${product.accent}66`,
   431→   431→                      }
   432→   432→                    : undefined
   433→   433→                }
   434→   434→              >
   435→   435→                {s}
   436→   436→                {isSelected && !soldOut && (
   437→   437→                  <Check className="absolute -right-1 -top-1 h-4 w-4 rounded-full bg-black p-0.5 text-white" />
   438→   438→                )}
   439→   439→                {!isSelected && lowStock && (
   440→   440→                  <span className="absolute -bottom-1 left-1/2 -translate-x-1/2 whitespace-nowrap rounded-full bg-amber-500/80 px-1 text-[8px] font-bold text-black">
   441→   441→                    {effectiveStock} rest.
   442→   442→                  </span>
   443→   443→                )}
   444→   444→              </button>
   445→   445→            );
   446→   446→          })}
   447→   447→        </div>
   448→   448→        {product.stock === 0 && (
   449→   449→          <p className="text-xs font-medium text-rose-400">
   450→   450→            Este modelo está temporariamente esgotado.
   451→   451→          </p>
   452→   452→        )}
   453→   453→        {size !== null && (() => {
   454→   454→          const sizeStockMap = product.sizeStock ?? {};
   455→   455→          const perSize = sizeStockMap[String(size)];
   456→   456→          const eff = perSize !== undefined ? perSize : product.stock;
   457→   457→          return eff <= 2 && eff > 0 ? (
   458→   458→            <p className="text-xs font-medium text-amber-300">
   459→   459→              Apenas {eff} unidade{eff === 1 ? "" : "s"} neste tamanho. Corra!
   460→   460→            </p>
   461→   461→          ) : null;
   462→   462→        })()}

   583→   583→

   585→   585→  const [author, setAuthor] = useState("");
   586→   586→  const [rating, setRating] = useState(0);
   587→   587→  const [hoverRating, setHoverRating] = useState(0);
   588→   588→  const [comment, setComment] = useState("");
