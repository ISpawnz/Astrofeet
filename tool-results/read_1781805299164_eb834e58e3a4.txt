   118→   118→function getInitial(name: string | undefined): string {
   119→   119→  if (!name) return "✦";
   120→   120→  const trimmed = name.trim();

   122→   122→  return trimmed[0]!.toUpperCase();

   125→   125→function itemsSummary(order: Order): string {
   126→   126→  const items = order.items;
   127→   127→  if (items.length === 0) return "Sem itens";
   128→   128→  const firstTwo = items.slice(0, 2).map((i) => `${i.name} (Tam ${i.size})`);
   129→   129→  const more = items.length - 2;
   130→   130→  const base = firstTwo.join(", ");
   131→   131→  if (more > 0) return `${base} +${more} mais`;
   132→   132→  return base;
   133→   133→}

   136→   136→  return order.items.reduce((n, i) => n + i.quantity, 0);
