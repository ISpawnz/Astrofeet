   164→function slugify(input: string): string {
   165→  return input
   166→    .toLowerCase()
   167→    .normalize("NFD")
   168→    .replace(/[\u0300-\u036f]/g, "")
   169→    .replace(/[^a-z0-9]+/g, "-")
   170→    .replace(/(^-|-$)+/g, "");
   171→}

   804→function emptyForm(): ProductFormState {
   805→  return {
   806→    name: "",
   807→    slug: "",
   808→    brand: "",
   809→    category: CATEGORIES[0],

   823→
   824→function formFromProduct(p: Product): ProductFormState {
   825→  const ss = p.sizeStock ?? {};
   826→  const sizeStock: Record<string, string> = {};
   827→  for (const s of p.sizes) {
   828→    const v = ss[String(s)];
   829→    sizeStock[String(s)] = v !== undefined ? String(v) : "";
   830→  }
   831→  return {
   832→    name: p.name,
   833→    slug: p.slug,
   834→    brand: p.brand,
   835→    category: p.category,
   836→    price: String(p.price),
   837→    stock: String(p.stock),
   838→    rating: String(p.rating),
   839→    accent: p.accent,
   840→    badge: p.badge ?? "",
   841→    description: p.description,
   842→    sizes: [...p.sizes].sort((a, b) => a - b),
   843→    sizeStock,
   844→    images: p.images.join(", "),
   845→    featured: p.featured,
   846→    bestSeller: p.bestSeller,
   847→  };
   848→}

   867→      setForm(editing ? formFromProduct(editing) : emptyForm());

   872→  function update<K extends keyof ProductFormState>(
   873→    key: K,
   874→    value: ProductFormState[K],

   876→    setForm((f) => ({ ...f, [key]: value }));
   877→  }
   878→
   879→  function updateName(v: string) {
   880→    setForm((f) => {
   881→      const next = { ...f, name: v };
   882→      if (!slugEdited) next.slug = slugify(v);

   887→  function toggleSize(s: number) {

   889→      const has = f.sizes.includes(s);

   892→        : [...f.sizes, s].sort((a, b) => a - b);
   893→      // Clean up sizeStock for removed sizes.
   894→      const sizeStock = { ...f.sizeStock };
   895→      if (has) delete sizeStock[String(s)];
   896→      return { ...f, sizes, sizeStock };
   897→    });
   898→  }

   900→  function setSizeStock(s: number, value: string) {

  1691→const COUPON_CODE_RE = /^[A-Z0-9]{3,20}$/;
  1692→
  1693→function CouponFormModal({
  1694→  open,

  1696→  onOpenChange,

  1703→  const [form, setForm] = useState<CouponFormState>(emptyCouponForm());

  1714→
  1715→  function update<K extends keyof CouponFormState>(
  1716→    key: K,
  1717→    value: CouponFormState[K],
  1718→  ) {
  1719→    setForm((f) => ({ ...f, [key]: value }));
  1720→    setErrors((e) => {
  1721→      if (!e[key]) return e;
  1722→      const next = { ...e };
  1723→      delete next[key];
  1724→      return next;
  1725→    });
  1726→  }
  1727→
  1728→  function onCodeChange(v: string) {
  1729→    const upper = v
  1730→      .toUpperCase()
  1731→      .replace(/[^A-Z0-9]/g, "")
  1732→      .slice(0, 20);
  1733→    update("code", upper);
  1734→  }
  1735→
  1736→  async function onSubmit(e: React.FormEvent) {
  1737→    e.preventDefault();
  1738→    if (saving) return;
  1739→
  1740→    const code = form.code.trim();
  1741→    const valueNum = Number(form.value);
  1742→    const minSubtotalNum = Number(form.minSubtotal) || 0;
  1743→    const description = form.description.trim();
  1744→
  1745→    const nextErrors: Record<string, string> = {};

  1749→      nextErrors.value = "Informe um valor maior que zero.";
  1750→    if (form.type === "percent" && valueNum > 100)
  1751→      nextErrors.value = "O percentual não pode passar de 100.";
  1752→    if (!description)
  1753→      nextErrors.description = "Descreva o cupom em poucas palavras.";
  1754→
  1755→    if (Object.keys(nextErrors).length > 0) {
  1756→      setErrors(nextErrors);
