    74→function paymentMethodInfo(method: string): {
    75→  label: string;
    76→  icon: React.ReactNode;
    77→} {
    78→  switch (method) {
    79→    case "card":
    80→      return { label: "Cartão de crédito", icon: <CreditCard className="h-4 w-4" /> };
    81→    case "pix":
    82→      return { label: "Pix", icon: <QrCode className="h-4 w-4" /> };
    83→    case "boleto":
    84→      return { label: "Boleto", icon: <Barcode className="h-4 w-4" /> };
    85→    default:
    86→      return { label: method || "Pagamento", icon: <CreditCard className="h-4 w-4" /> };
    87→  }

    90→function getInitial(name: string | undefined): string {
    91→  if (!name) return "✦";
    92→  const trimmed = name.trim();
    93→  if (!trimmed) return "✦";
    94→  return trimmed[0]!.toUpperCase();
    95→}

    97→function itemsSummary(order: Order): string {
    98→  const items = order.items;
    99→  if (items.length === 0) return "Sem itens";
   100→  const firstTwo = items.slice(0, 2).map((i) => `${i.name} (Tam ${i.size})`);
   101→  const more = items.length - 2;
   102→  const base = firstTwo.join(", ");
   103→  if (more > 0) return `${base} +${more} mais`;
   104→  return base;
   105→}
   106→
   107→function itemCount(order: Order): number {
   108→  return order.items.reduce((n, i) => n + i.quantity, 0);
   109→}

   111→function isCancelled(status: string): boolean {
   112→  return status === "cancelled";

   743→function ProfileTab({ user }: { user: PublicUser }) {
   744→  const navigate = useUIStore((s) => s.navigate);
   745→  const setUser = useAuthStore((s) => s.setUser);
   746→  const roleLabel = user.role === "admin" ? "Comando" : "Explorador";
   747→  const roleAccent =
   748→    user.role === "admin" ? "var(--neon-magenta)" : "var(--neon-cyan)";
   749→
   750→  // Editable name state
   751→  const [editingName, setEditingName] = useState(false);
   752→  const [nameValue, setNameValue] = useState(user.name);
   753→  const [savingName, setSavingName] = useState(false);
   754→
   755→  useEffect(() => {
   756→    setNameValue(user.name);
   757→  }, [user.name]);
   758→
   759→  async function saveName() {
   760→    const trimmed = nameValue.trim();
   761→    if (trimmed.length < 2) {
   762→      toast.error("Nome precisa ter ao menos 2 caracteres.");
   763→      return;
   764→    }
   765→    if (trimmed === user.name) {
   766→      setEditingName(false);
   767→      return;
   768→    }
   769→    setSavingName(true);
   770→    try {
   771→      const updated = await api.updateProfile(trimmed);
   772→      setUser(updated);
   773→      toast.success("Nome atualizado com sucesso!");
   774→      setEditingName(false);
   775→    } catch (err) {
   776→      toast.error(
   777→        err instanceof Error ? err.message : "Não foi possível salvar.",
   778→      );
   779→    } finally {
   780→      setSavingName(false);
   781→    }
   782→  }
   783→
   784→  return (
   785→    <div className="grid gap-5 lg:grid-cols-2">
   786→      {/* Read-only profile */}
   787→      <motion.div
   788→        initial={{ opacity: 0, y: 16 }}
   789→        animate={{ opacity: 1, y: 0 }}
   790→        transition={{ duration: 0.45 }}
   791→        className="glass rounded-3xl p-6"
   792→      >
   793→        <h2 className="flex items-center gap-2 text-sm font-bold uppercase tracking-wider text-muted-foreground">
   794→          <User className="h-4 w-4 text-[var(--neon-cyan)]" />
   795→          Meus dados
   796→        </h2>
   797→        <div className="mt-4 space-y-2.5">
   798→          {/* Editable name row */}
   799→          <div className="flex items-center gap-3 rounded-2xl border border-white/5 bg-white/[0.03] p-3">
   800→            <span
   801→              className="flex h-9 w-9 shrink-0 items-center justify-center rounded-xl"
   802→              style={{ background: "color-mix(in oklab, var(--neon-cyan) 12%, transparent)" }}
   803→            >
   804→              <User className="h-4 w-4 text-[var(--neon-cyan)]" />
   805→            </span>
   806→            <div className="flex-1">
   807→              <p className="text-[11px] uppercase tracking-wider text-muted-foreground">
   808→                Nome
   809→              </p>
   810→              {editingName ? (
   811→                <div className="mt-1 flex items-center gap-2">
   812→                  <input
   813→                    autoFocus
   814→                    value={nameValue}
   815→                    onChange={(e) => setNameValue(e.target.value)}
   816→                    onKeyDown={(e) => {
   817→                      if (e.key === "Enter") saveName();
   818→                      if (e.key === "Escape") {
   819→                        setEditingName(false);
   820→                        setNameValue(user.name);
   821→                      }
