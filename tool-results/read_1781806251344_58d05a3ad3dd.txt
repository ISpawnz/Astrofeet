   234→function Info({
   235→  product,
   236→  reviews,
   237→}: {
   238→  product: Product;
   239→  reviews: Review[];
   240→}) {
   241→  const add = useCartStore((s) => s.add);
   242→  const openCart = useCartStore((s) => s.open);
   243→  const navigate = useUIStore((s) => s.navigate);
   244→  const openSizeGuide = useUIStore((s) => s.openSizeGuide);
   245→  const toggleWishlist = useWishlistStore((s) => s.toggleProduct);
   246→  const inWishlist = useWishlistStore((s) => s.has(product.id));
   247→  const [heartBump, setHeartBump] = useState(false);
   248→
   249→  const [size, setSize] = useState<number | null>(null);
   250→  const [qty, setQty] = useState(1);
   251→
   252→  // Per-size-aware max quantity (fall back to global stock when sizeStock absent).
   253→  const sizeStockMap = product.sizeStock ?? {};
   254→  const selectedSizeStock =
   255→    size !== null ? sizeStockMap[String(size)] : undefined;
   256→  const effectiveMax =
   257→    selectedSizeStock !== undefined ? selectedSizeStock : product.stock;
   258→  const maxQty = Math.max(1, Math.min(effectiveMax, 99));
   259→  const installment = product.price / 10;
   260→  const freeShipping = product.price >= FREE_SHIPPING_THRESHOLD;
   261→  const reviewCount = reviews.length;
   262→
   263→  // Reset qty if it exceeds the newly-selected size's stock.
   264→  useEffect(() => {
   265→    // eslint-disable-next-line react-hooks/set-state-in-effect
   266→    if (qty > maxQty) setQty(Math.max(1, maxQty));
   267→  }, [maxQty, qty]);
   268→
   269→  function selectSize(s: number) {
   270→    setSize(s);
   271→    setQty(1);
   272→  }
   273→
   274→  function requireSize(): boolean {
   275→    if (size === null) {
   276→      toast.error("Selecione um tamanho para continuar.");
   277→      return false;
   278→    }
   279→    return true;
   280→  }
   281→
   282→  function handleAdd() {
   283→    if (!requireSize()) return;
   284→    add(product, size as number, qty);
   285→    toast.success(`${product.name} adicionado ao carrinho`);
   286→    openCart();
   287→  }
   288→
   289→  function handleBuyNow() {
   290→    if (!requireSize()) return;
   291→    add(product, size as number, qty);
   292→    navigate("checkout");
   293→  }
   294→
   295→  function handleWishlist() {
   296→    toggleWishlist(product);
   297→    setHeartBump(true);
   298→    setTimeout(() => setHeartBump(false), 450);
   299→    toast.success(
   300→      inWishlist
   301→        ? `${product.name} saiu da sua lista`
   302→        : `${product.name} salvo na lista de desejos`,
   303→    );
   304→  }
   305→
   306→  return (
   307→    <motion.div
   308→      initial={{ opacity: 0, y: 20 }}
   309→      animate={{ opacity: 1, y: 0 }}
   310→      transition={{ duration: 0.5, delay: 0.1 }}
   311→      className="flex flex-col gap-5"
   312→    >
   313→      {/* Brand + wishlist */}
   314→      <div className="flex items-start justify-between gap-3">
   315→        <p className="text-xs font-semibold uppercase tracking-[0.2em] text-muted-foreground">
   316→          {product.brand}
   317→        </p>
   318→        <button
   319→          onClick={handleWishlist}
   320→          className={cn(
   321→            "flex h-10 w-10 shrink-0 items-center justify-center rounded-full border transition-all hover:scale-110",
   322→            inWishlist
   323→              ? "border-[var(--neon-magenta)]/50 bg-[var(--neon-magenta)]/20 text-[var(--neon-magenta)]"
   324→              : "border-white/10 bg-white/5 text-foreground/70 hover:text-foreground",
   325→          )}
   326→          aria-label={
   327→            inWishlist ? "Remover da lista de desejos" : "Salvar na lista de desejos"
   328→          }
   329→          aria-pressed={inWishlist}
   330→        >
   331→          <Heart
   332→            className={cn("h-5 w-5", heartBump && "animate-heartbeat", inWishlist && "fill-current")}
   333→          />
   334→        </button>
   335→      </div>
   336→
   337→      {/* Name */}
   338→      <h1 className="text-3xl font-black leading-tight sm:text-4xl">
   339→        {product.name}
   340→      </h1>
   341→
   342→      {/* Rating */}
   343→      <div className="flex items-center gap-2 text-sm">
   344→        <Stars value={product.rating} size={16} />
   345→        <span className="font-semibold">{product.rating.toFixed(1)}</span>
   346→        <span className="text-muted-foreground">
   347→          · {reviewCount}{" "}
   348→          {reviewCount === 1 ? "avaliação" : "avaliações"}
   349→        </span>
   350→      </div>
   351→
   352→      {/* Price */}
   353→      <div className="space-y-1">
   354→        <p className="text-3xl font-bold">{formatPrice(product.price)}</p>
   355→        <p className="text-sm text-muted-foreground">
   356→          ou 10x de{" "}
   357→          <span className="font-semibold text-foreground">
   358→            {formatPrice(installment)}
   359→          </span>{" "}
   360→          sem juros
   361→        </p>
   362→        <p
   363→          className={cn(
   364→            "text-xs font-medium",
   365→            freeShipping ? "text-[var(--neon-lime)]" : "text-muted-foreground",
   366→          )}
   367→        >
   368→          {freeShipping
   369→            ? "🚀 Frete grátis acima de R$300 — liberado!"
   370→            : "Frete grátis em pedidos acima de R$300"}
   371→        </p>
   372→      </div>
   373→
   374→      <Separator className="bg-white/10" />
   375→
   376→      {/* Short description */}
   377→      <p className="text-sm leading-relaxed text-muted-foreground">
   378→        {product.description.split("\n")[0]?.slice(0, 220) ??
   379→          "Modelo exclusivo Astrofeet, feito para quem anda entre estrelas."}
   380→      </p>
   381→
   382→      {/* Size selector */}
   383→      <div className="space-y-2">
   384→        <div className="flex items-center justify-between">
   385→          <label className="text-sm font-semibold">Tamanho</label>
   386→          <button
   387→            onClick={openSizeGuide}
   388→            className="inline-flex items-center gap-1.5 text-xs font-medium text-[var(--neon-cyan)] transition hover:underline"
   389→          >
   390→            <Ruler className="h-3.5 w-3.5" />
   391→            Guia de medidas
   392→          </button>
   393→        </div>
   394→        <div className="flex flex-wrap gap-2">
   395→          {product.sizes.map((s) => {
   396→            const isSelected = size === s;
   397→            // Per-size stock: prefer sizeStock map, fall back to global stock.
   398→            const sizeStockMap = product.sizeStock ?? {};
   399→            const perSize = sizeStockMap[String(s)];
   400→            const effectiveStock =
   401→              perSize !== undefined ? perSize : product.stock;
   402→            const soldOut = effectiveStock <= 0;
   403→            const lowStock = !soldOut && effectiveStock <= 2;
   404→            return (
   405→              <button
   406→                key={s}
   407→                onClick={() => !soldOut && selectSize(s)}
   408→                disabled={soldOut}
   409→                aria-pressed={isSelected}
   410→                title={
   411→                  soldOut
   412→                    ? `Tamanho ${s} esgotado`
   413→                    : lowStock
   414→                      ? `Últimas ${effectiveStock} unidades`
   415→                      : `Tamanho ${s}`
   416→                }
   417→                className={cn(
   418→                  "relative h-12 min-w-14 rounded-xl border px-3 text-sm font-bold transition",
   419→                  soldOut && "cursor-not-allowed opacity-40 line-through",
   420→                  isSelected && !soldOut
   421→                    ? "border-transparent text-black"
   422→                    : lowStock
   423→                      ? "border-amber-500/40 bg-amber-500/5 text-amber-200 hover:border-amber-500/60"
   424→                      : "border-white/10 bg-white/[0.03] hover:border-white/30 hover:bg-white/[0.06]",
   425→                )}
   426→                style={
   427→                  isSelected && !soldOut
   428→                    ? {
   429→                        background: product.accent,
   430→                        boxShadow: `0 0 18px ${product.accent}66`,
   431→                      }
   432→                    : undefined
   433→                }
   434→              >
   435→                {s}
   436→                {isSelected && !soldOut && (
   437→                  <Check className="absolute -right-1 -top-1 h-4 w-4 rounded-full bg-black p-0.5 text-white" />
   438→                )}
   439→                {!isSelected && lowStock && (
   440→                  <span className="absolute -bottom-1 left-1/2 -translate-x-1/2 whitespace-nowrap rounded-full bg-amber-500/80 px-1 text-[8px] font-bold text-black">
   441→                    {effectiveStock} rest.
   442→                  </span>
   443→                )}
   444→              </button>
   445→            );
   446→          })}
   447→        </div>
   448→        {product.stock === 0 && (
   449→          <p className="text-xs font-medium text-rose-400">
   450→            Este modelo está temporariamente esgotado.
   451→          </p>
   452→        )}
   453→        {size !== null && (() => {
   454→          const sizeStockMap = product.sizeStock ?? {};
   455→          const perSize = sizeStockMap[String(size)];
   456→          const eff = perSize !== undefined ? perSize : product.stock;
   457→          return eff <= 2 && eff > 0 ? (
   458→            <p className="text-xs font-medium text-amber-300">
   459→              Apenas {eff} unidade{eff === 1 ? "" : "s"} neste tamanho. Corra!
   460→            </p>
   461→          ) : null;
   462→        })()}
   463→      </div>
   464→
   465→      {/* Quantity + Stock */}
   466→      <div className="flex flex-wrap items-center gap-4">
   467→        <div className="flex items-center gap-1 rounded-full border border-white/10 bg-white/[0.03] p-1">
   468→          <button
   469→            onClick={() => setQty((q) => Math.max(1, q - 1))}
   470→            disabled={qty <= 1}
   471→            aria-label="Diminuir quantidade"
   472→            className="flex h-9 w-9 items-center justify-center rounded-full transition hover:bg-white/10 disabled:opacity-40"
   473→          >
   474→            <Minus className="h-4 w-4" />
   475→          </button>
   476→          <span className="w-8 text-center text-sm font-bold">{qty}</span>
   477→          <button
   478→            onClick={() => setQty((q) => Math.min(maxQty, q + 1))}
   479→            disabled={qty >= maxQty}
   480→            aria-label="Aumentar quantidade"
   481→            className="flex h-9 w-9 items-center justify-center rounded-full transition hover:bg-white/10 disabled:opacity-40"
   482→          >
   483→            <Plus className="h-4 w-4" />
   484→          </button>
   485→        </div>
   486→
   487→        {product.stock > 5 ? (
   488→          <span className="inline-flex items-center gap-1.5 text-xs font-medium text-emerald-400">
   489→            <span className="h-2 w-2 rounded-full bg-emerald-400" />
   490→            Em estoque
   491→          </span>
   492→        ) : product.stock > 0 ? (
   493→          <span className="inline-flex items-center gap-1.5 text-xs font-medium text-amber-300">
   494→            <span className="h-2 w-2 rounded-full bg-amber-400" />
   495→            Últimas {product.stock} unidades
   496→          </span>
   497→        ) : (
   498→          <span className="inline-flex items-center gap-1.5 text-xs font-medium text-rose-400">
   499→            <span className="h-2 w-2 rounded-full bg-rose-400" />
   500→            Esgotado
   501→          </span>
   502→        )}
   503→      </div>
   504→
   505→      {/* CTAs */}
   506→      <div className="grid gap-2.5 sm:grid-cols-[1fr_auto]">
   507→        <Button
   508→          onClick={handleAdd}
   509→          disabled={product.stock === 0}
   510→          size="lg"
   511→          className="h-12 rounded-full bg-gradient-to-r from-[var(--neon-cyan)] to-[var(--neon-violet)] text-base font-bold text-black transition hover:opacity-90 disabled:opacity-40"
   512→        >
   513→          <ShoppingCart className="mr-2 h-5 w-5" />
   514→          Adicionar ao carrinho
   515→        </Button>
   516→        <Button
   517→          onClick={handleBuyNow}
   518→          disabled={product.stock === 0}
   519→          size="lg"
   520→          variant="outline"
   521→          className="h-12 rounded-full border-white/20 bg-white/[0.03] px-6 text-base font-bold text-white backdrop-blur hover:border-white/40 hover:bg-white/[0.08] disabled:opacity-40"
   522→        >
   523→          Comprar agora
   524→        </Button>
   525→      </div>
   526→
   527→      {/* Trust row */}
   528→      <TrustRow />
   529→    </motion.div>
   530→  );

   537→function InteractiveStars({
   538→  value,

   540→}: {
   541→  value: number;
   542→  onChange: (v: number) => void;
   543→}) {
   544→  const [hover, setHover] = useState(0);
   545→  const display = hover || value;
   546→
   547→  return (
   548→    <div className="flex items-center gap-1">
