   141→   141→function slugify(input: string): string {
   142→   142→  return input
   143→   143→    .toLowerCase()
   144→   144→    .normalize("NFD")
   145→   145→    .replace(/[\u0300-\u036f]/g, "")
   146→   146→    .replace(/[^a-z0-9]+/g, "-")
   147→   147→    .replace(/(^-|-$)+/g, "");
   148→   148→}
