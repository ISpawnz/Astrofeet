   164→   164→function slugify(input: string): string {
   165→   165→  return input
   166→   166→    .toLowerCase()
   167→   167→    .normalize("NFD")
   168→   168→    .replace(/[\u0300-\u036f]/g, "")
   169→   169→    .replace(/[^a-z0-9]+/g, "-")
   170→   170→    .replace(/(^-|-$)+/g, "");
   171→   171→}

   824→   824→function formFromProduct(p: Product): ProductFormState {
   825→   825→  const ss = p.sizeStock ?? {};
   826→   826→  const sizeStock: Record<string, string> = {};
   827→   827→  for (const s of p.sizes) {

   830→   830→  }
   831→   831→  return {
   832→   832→    name: p.name,
   833→   833→    slug: p.slug,
   834→   834→    brand: p.brand,

   840→   840→    badge: p.badge ?? "",
   841→   841→    description: p.description,
   842→   842→    sizes: [...p.sizes].sort((a, b) => a - b),
   843→   843→    sizeStock,
   844→   844→    images: p.images.join(", "),
   845→   845→    featured: p.featured,

   854→   854→}: {

   856→   856→  editing: Product | null;

   862→   862→  const [saving, setSaving] = useState(false);
   863→   863→
   864→   864→  // Sync form whenever modal opens or editing target changes
   865→   865→  useEffect(() => {
   866→   866→    if (open) {
   867→   867→      setForm(editing ? formFromProduct(editing) : emptyForm());

   872→   872→  function update<K extends keyof ProductFormState>(
   873→   873→    key: K,

   879→   879→  function updateName(v: string) {

   883→   883→      return next;
   884→   884→    });
   885→   885→  }
   886→   886→
   887→   887→  function toggleSize(s: number) {
   888→   888→    setForm((f) => {
   889→   889→      const has = f.sizes.includes(s);
   890→   890→      const sizes = has
   891→   891→        ? f.sizes.filter((x) => x !== s)
   892→   892→        : [...f.sizes, s].sort((a, b) => a - b);
   893→   893→      // Clean up sizeStock for removed sizes.
   894→   894→      const sizeStock = { ...f.sizeStock };
   895→   895→      if (has) delete sizeStock[String(s)];
   896→   896→      return { ...f, sizes, sizeStock };
   897→   897→    });
   898→   898→  }

   900→   900→  function setSizeStock(s: number, value: string) {
   901→   901→    setForm((f) => ({
   902→   902→      ...f,
   903→   903→      sizeStock: { ...f.sizeStock, [String(s)]: value },
   904→   904→    }));
   905→   905→  }
   906→   906→
   907→   907→  async function onSubmit(e: React.FormEvent) {
   908→   908→    e.preventDefault();
   909→   909→    if (saving) return;
   910→   910→
   911→   911→    const name = form.name.trim();
   912→   912→    const brand = form.brand.trim();
   913→   913→    const category = form.category.trim();
   914→   914→    const priceNum = Number(form.price);
   915→   915→    const stockNum = parseInt(form.stock || "0", 10);
   916→   916→    const ratingNum = Math.min(5, Math.max(0, Number(form.rating) || 0));
   917→   917→
   918→   918→    if (!name) return toast.error("Dê um nome ao produto.");
   919→   919→    if (!brand) return toast.error("Informe a marca.");
   920→   920→    if (!category) return toast.error("Informe a categoria.");
   921→   921→    if (!Number.isFinite(priceNum) || priceNum < 0)
   922→   922→      return toast.error("Informe um preço válido.");
   923→   923→    if (form.sizes.length === 0)
   924→   924→      return toast.error("Selecione ao menos um tamanho.");
   925→   925→
   926→   926→    const slug = (form.slug.trim() || slugify(name)).toLowerCase();
   927→   927→    const imagesRaw = form.images
   928→   928→      .split(",")
   929→   929→      .map((s) => s.trim())

   933→   933→

   937→   937→      const raw = form.sizeStock[String(s)];
   938→   938→      const v = raw === "" ? undefined : parseInt(raw || "0", 10);
   939→   939→      if (v !== undefined && Number.isFinite(v) && v >= 0)
   940→   940→        sizeStockMap[String(s)] = v;
   941→   941→    }
   942→   942→
   943→   943→    const body: Partial<Product> = {
   944→   944→      name,
   945→   945→      slug,
   946→   946→      brand,
   947→   947→      category,
   948→   948→      price: priceNum,
   949→   949→      stock: Number.isFinite(stockNum) ? stockNum : 0,
   950→   950→      rating: ratingNum,
   951→   951→      accent: form.accent,
   952→   952→      badge: form.badge ? form.badge : null,
   953→   953→      description: form.description.trim(),
   954→   954→      sizes: form.sizes,

   969→   969→        toast.success(`${name} adicionado ao catálogo.`);
   970→   970→      }
   971→   971→      await queryClient.invalidateQueries({ queryKey: ["products"] });
   972→   972→      await queryClient.invalidateQueries({ queryKey: ["admin-metrics"] });
   973→   973→      onOpenChange(false);
   974→   974→    } catch (err) {
   975→   975→      toast.error(
   976→   976→        err instanceof Error
   977→   977→          ? err.message
   978→   978→          : "Não foi possível salvar o produto.",
   979→   979→      );
   980→   980→    } finally {
   981→   981→      setSaving(false);
   982→   982→    }
   983→   983→  }
   984→   984→
   985→   985→  return (
   986→   986→    <Dialog open={open} onOpenChange={onOpenChange}>
