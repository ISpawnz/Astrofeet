   102→function paymentMethodInfo(method: string): {
   103→  label: string;
   104→  icon: React.ReactNode;
   105→} {
   106→  switch (method) {
   107→    case "card":
   108→      return { label: "Cartão de crédito", icon: <CreditCard className="h-4 w-4" /> };
   109→    case "pix":
   110→      return { label: "Pix", icon: <QrCode className="h-4 w-4" /> };
   111→    case "boleto":
   112→      return { label: "Boleto", icon: <Barcode className="h-4 w-4" /> };
   113→    default:
   114→      return { label: method || "Pagamento", icon: <CreditCard className="h-4 w-4" /> };
   115→  }
   116→}

   118→function getInitial(name: string | undefined): string {
   119→  if (!name) return "✦";
   120→  const trimmed = name.trim();
   121→  if (!trimmed) return "✦";
   122→  return trimmed[0]!.toUpperCase();
   123→}
   124→
   125→function itemsSummary(order: Order): string {
   126→  const items = order.items;
   127→  if (items.length === 0) return "Sem itens";
   128→  const firstTwo = items.slice(0, 2).map((i) => `${i.name} (Tam ${i.size})`);
   129→  const more = items.length - 2;
   130→  const base = firstTwo.join(", ");
   131→  if (more > 0) return `${base} +${more} mais`;
   132→  return base;
   133→}
   134→
   135→function itemCount(order: Order): number {
   136→  return order.items.reduce((n, i) => n + i.quantity, 0);
   137→}
   138→
   139→function isCancelled(status: string): boolean {
   140→  return status === "cancelled";

   771→function ProfileTab({ user }: { user: PublicUser }) {

   773→  const setUser = useAuthStore((s) => s.setUser);
   774→  const roleLabel = user.role === "admin" ? "Comando" : "Explorador";
   775→  const roleAccent =
   776→    user.role === "admin" ? "var(--neon-magenta)" : "var(--neon-cyan)";
   777→
   778→  // Editable name state
   779→  const [editingName, setEditingName] = useState(false);
   780→  const [nameValue, setNameValue] = useState(user.name);
   781→  const [savingName, setSavingName] = useState(false);
   782→
   783→  // Password change state
   784→  const [currentPassword, setCurrentPassword] = useState("");
   785→  const [newPassword, setNewPassword] = useState("");
   786→  const [confirmPassword, setConfirmPassword] = useState("");
   787→  const [savingPassword, setSavingPassword] = useState(false);
   788→  const [showCurrent, setShowCurrent] = useState(false);
   789→  const [showNew, setShowNew] = useState(false);
   790→  const [showConfirm, setShowConfirm] = useState(false);
   791→
   792→  useEffect(() => {
   793→    setNameValue(user.name);
   794→  }, [user.name]);
   795→
   796→  async function saveName() {
   797→    const trimmed = nameValue.trim();
   798→    if (trimmed.length < 2) {
   799→      toast.error("Nome precisa ter ao menos 2 caracteres.");
   800→      return;
   801→    }
   802→    if (trimmed === user.name) {
   803→      setEditingName(false);
   804→      return;
   805→    }
   806→    setSavingName(true);
   807→    try {
   808→      const updated = await api.updateProfile(trimmed);
   809→      setUser(updated);
   810→      toast.success("Nome atualizado com sucesso!");
   811→      setEditingName(false);
   812→    } catch (err) {
   813→      toast.error(
   814→        err instanceof Error ? err.message : "Não foi possível salvar.",
   815→      );
   816→    } finally {
   817→      setSavingName(false);
   818→    }
   819→  }
   820→
   821→  function resetPasswordFields() {
   822→    setCurrentPassword("");
   823→    setNewPassword("");
   824→    setConfirmPassword("");

   826→
   827→  async function handleChangePassword() {
   828→    if (!currentPassword) {
   829→      toast.error("Informe sua senha atual.");
   830→      return;
   831→    }
   832→    if (newPassword.length < 6) {
   833→      toast.error("A nova senha precisa ter ao menos 6 caracteres.");
   834→      return;
   835→    }
   836→    if (newPassword === currentPassword) {
   837→      toast.error("A nova senha precisa ser diferente da atual.");
   838→      return;
   839→    }
   840→    if (newPassword !== confirmPassword) {
   841→      toast.error("As senhas não coincidem.");
   842→      return;
   843→    }
   844→    setSavingPassword(true);
   845→    try {
   846→      await api.changePassword(currentPassword, newPassword);
   847→      toast.success("Senha atualizada com sucesso.");
   848→      resetPasswordFields();
   849→    } catch (err) {
   850→      toast.error(
   851→        err instanceof Error
   852→          ? err.message
   853→          : "Não foi possível atualizar a senha agora.",
   854→      );
   855→    } finally {
   856→      setSavingPassword(false);
   857→    }
   858→  }
   859→
   860→  return (
   861→    <div className="space-y-5">
   862→      {/* Read-only profile */}
