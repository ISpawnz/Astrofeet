   145→function slugify(input: string): string {
   146→  return input
   147→    .toLowerCase()
   148→    .normalize("NFD")
   149→    .replace(/[\u0300-\u036f]/g, "")
   150→    .replace(/[^a-z0-9]+/g, "-")
   151→    .replace(/(^-|-$)+/g, "");
   152→}

   561→
   562→function OrdersTable() {
   563→  const queryClient = useQueryClient();
   564→  const [query, setQuery] = useState("");
   565→  const [updating, setUpdating] = useState<string | null>(null);
   566→
   567→  const { data, isLoading, isError } = useQuery({
   568→    queryKey: ["orders"],
   569→    queryFn: () => api.listOrders(),
   570→  });
   571→
   572→  const filtered = useMemo(() => {
   573→    if (!data) return [];
   574→    const q = query.trim().toLowerCase();
   575→    if (!q) return data;
   576→    return data.filter(
   577→      (o) =>
   578→        o.code.toLowerCase().includes(q) ||
   579→        o.customer.name.toLowerCase().includes(q),
   580→    );
   581→  }, [data, query]);
   582→
   583→  async function onStatusChange(order: Order, status: string) {
   584→    setUpdating(order.id);
   585→    try {
   586→      await api.updateOrderStatus(order.id, status);
   587→      await Promise.all([
   588→        queryClient.invalidateQueries({ queryKey: ["orders"] }),
   589→        queryClient.invalidateQueries({ queryKey: ["admin-metrics"] }),
   590→      ]);
   591→      toast.success(`Pedido ${order.code} atualizado para "${orderStatusLabel(status)}".`);
   592→    } catch (err) {
   593→      toast.error(
   594→        err instanceof Error ? err.message : "Não foi possível atualizar o status.",
   595→      );
   596→    } finally {
   597→      setUpdating(null);
   598→    }
   599→  }
   600→
   601→  if (isError) {
   602→    return (
   603→      <div className="glass rounded-2xl border border-white/10 p-8 text-center text-sm text-muted-foreground">
   604→        Não foi possível carregar os pedidos. Tente novamente em instantes.
   605→      </div>
   606→    );
   607→  }
   608→
   609→  return (
   610→    <div className="glass rounded-2xl border border-white/10 p-4 sm:p-5">
   611→      <div className="mb-4 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   612→        <div>
   613→          <h3 className="font-semibold">Todos os pedidos</h3>
   614→          <p className="text-xs text-muted-foreground">
   615→            Atualize o status de cada pedido em tempo real.
   616→          </p>

   735→interface ProductFormState {
   736→  name: string;

   753→function emptyForm(): ProductFormState {
   754→  return {
   755→    name: "",
   756→    slug: "",
   757→    brand: "",
   758→    category: CATEGORIES[0],
   759→    price: "",
   760→    stock: "0",
   761→    rating: "5",
   762→    accent: ACCENT_PRESETS[0].value,
   763→    badge: "",
   764→    description: "",
   765→    sizes: [...ALL_SIZES],
   766→    sizeStock: {},
   767→    images: "",
   768→    featured: false,
   769→    bestSeller: false,
   770→  };
   771→}
   772→
   773→function formFromProduct(p: Product): ProductFormState {
   774→  const ss = p.sizeStock ?? {};
   775→  const sizeStock: Record<string, string> = {};
   776→  for (const s of p.sizes) {
   777→    const v = ss[String(s)];
   778→    sizeStock[String(s)] = v !== undefined ? String(v) : "";
   779→  }
   780→  return {
   781→    name: p.name,
   782→    slug: p.slug,
   783→    brand: p.brand,
   784→    category: p.category,
   785→    price: String(p.price),
   786→    stock: String(p.stock),
   787→    rating: String(p.rating),
   788→    accent: p.accent,
   789→    badge: p.badge ?? "",
   790→    description: p.description,
   791→    sizes: [...p.sizes].sort((a, b) => a - b),
   792→    sizeStock,
   793→    images: p.images.join(", "),
   794→    featured: p.featured,
   795→    bestSeller: p.bestSeller,
   796→  };
   797→}

  1598→interface CouponFormState {

  1608→function emptyCouponForm(): CouponFormState {

  1610→    code: "",

  1620→function formFromCoupon(c: Coupon): CouponFormState {
  1621→  let expiresAt = "";
  1622→  if (c.expiresAt) {
  1623→    try {
  1624→      expiresAt = new Date(c.expiresAt).toISOString().slice(0, 10);
  1625→    } catch {
  1626→      expiresAt = "";

  1628→  }
  1629→  return {
  1630→    code: c.code,
  1631→    type: c.type,
  1632→    value: String(c.value),
  1633→    minSubtotal: String(c.minSubtotal ?? 0),
  1634→    description: c.description ?? "",
  1635→    expiresAt,
  1636→    active: c.active,

  1638→}
  1639→
  1640→const COUPON_CODE_RE = /^[A-Z0-9]{3,20}$/;
  1641→
  1642→function CouponFormModal({
  1643→  open,
  1644→  editing,
  1645→  onOpenChange,

  1649→  onOpenChange: (v: boolean) => void;

  1652→  const [form, setForm] = useState<CouponFormState>(emptyCouponForm());
  1653→  const [errors, setErrors] = useState<Record<string, string>>({});

  1655→
  1656→  // Sync form whenever modal opens or editing target changes
  1657→  useEffect(() => {
  1658→    if (open) {
  1659→      setForm(editing ? formFromCoupon(editing) : emptyCouponForm());
  1660→      setErrors({});

  1663→
  1664→  function update<K extends keyof CouponFormState>(
  1665→    key: K,
  1666→    value: CouponFormState[K],
  1667→  ) {
  1668→    setForm((f) => ({ ...f, [key]: value }));
  1669→    setErrors((e) => {
  1670→      if (!e[key]) return e;
  1671→      const next = { ...e };
  1672→      delete next[key];
  1673→      return next;
  1674→    });
  1675→  }
  1676→
  1677→  function onCodeChange(v: string) {
  1678→    const upper = v
  1679→      .toUpperCase()
  1680→      .replace(/[^A-Z0-9]/g, "")
  1681→      .slice(0, 20);
  1682→    update("code", upper);
  1683→  }
  1684→
  1685→  async function onSubmit(e: React.FormEvent) {
  1686→    e.preventDefault();

  1688→
  1689→    const code = form.code.trim();
  1690→    const valueNum = Number(form.value);
  1691→    const minSubtotalNum = Number(form.minSubtotal) || 0;
  1692→    const description = form.description.trim();

  1694→    const nextErrors: Record<string, string> = {};

  1703→
  1704→    if (Object.keys(nextErrors).length > 0) {
  1705→      setErrors(nextErrors);
  1706→      toast.error("Verifique os campos destacados.");
  1707→      return;
  1708→    }
  1709→
  1710→    const expiresAt = form.expiresAt
  1711→      ? new Date(`${form.expiresAt}T23:59:59`).toISOString()
  1712→      : null;
  1713→
  1714→    setSaving(true);
  1715→    try {
  1716→      if (editing) {
  1717→        await api.updateCoupon(editing.id, {
  1718→          code,
  1719→          value: valueNum,
  1720→          minSubtotal: minSubtotalNum,
  1721→          description,
  1722→          active: form.active,

  1728→          code,
  1729→          type: form.type,
  1730→          value: valueNum,
  1731→          minSubtotal: minSubtotalNum,
  1732→          description,
  1733→          active: form.active,
  1734→          expiresAt,
  1735→        });
  1736→        toast.success(`Cupom ${code} criado com sucesso.`);
  1737→      }
  1738→      await queryClient.invalidateQueries({ queryKey: ["coupons"] });
  1739→      onOpenChange(false);
  1740→    } catch (err) {
  1741→      toast.error(
  1742→        err instanceof Error
  1743→          ? err.message
  1744→          : "Não foi possível salvar o cupom.",
  1745→      );
  1746→    } finally {
  1747→      setSaving(false);
  1748→    }
  1749→  }
  1750→
