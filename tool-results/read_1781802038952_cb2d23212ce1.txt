    46→type PaymentMethod = "card" | "pix" | "boleto";
    47→
    48→interface FormState {
    49→  name: string;
    50→  email: string;
    51→  phone: string;
    52→  cep: string;
    53→  street: string;
    54→  number: string;
    55→  complement: string;
    56→  district: string;
    57→  city: string;
    58→  state: string;
    59→  payment: PaymentMethod;

    67→
    68→const inputClass =
    69→  "h-11 rounded-xl border border-white/10 bg-white/5 px-4 text-sm outline-none transition-colors placeholder:text-muted-foreground/60 focus:border-[var(--neon-cyan)] focus-visible:ring-0 focus-visible:ring-offset-0";
    70→
    71→function FieldError({ message }: { message?: string }) {
    72→  if (!message) return null;
    73→  return (
    74→    <p className="mt-1.5 text-xs text-rose-400" role="alert">
    75→      {message}
    76→    </p>
    77→  );
    78→}
    79→
    80→export function CheckoutView() {
    81→  const { items, subtotal, clear } = useCartStore();
    82→  const navigate = useUIStore((s) => s.navigate);
    83→  const user = useAuthStore((s) => s.user);
    84→  const setLastOrder = useCheckoutStore((s) => s.setLastOrder);
    85→
    86→  const [form, setForm] = useState<FormState>({
    87→    name: "",
    88→    email: "",
    89→    phone: "",
    90→    cep: "",
    91→    street: "",
    92→    number: "",
    93→    complement: "",
    94→    district: "",
    95→    city: "",
    96→    state: "",
    97→    payment: "card",
    98→    cardNumber: "",
    99→    cardName: "",
   100→    cardExpiry: "",
   101→    cardCVV: "",
   102→  });
   103→  const [errors, setErrors] = useState<Errors>({});
   104→  const [submitting, setSubmitting] = useState(false);
   105→
   106→  // Coupon state
   107→  const [couponInput, setCouponInput] = useState("");
   108→  const [couponState, setCouponState] = useState<{
   109→    status: "idle" | "validating" | "applied" | "error";
   110→    code?: string;
   111→    description?: string;
   112→    discount?: number;
   113→    message?: string;
   114→  }>({ status: "idle" });
   115→
   116→  // Prefill from authenticated user
   117→  useEffect(() => {
   118→    if (user) {
   119→      setForm((f) => ({
   120→        ...f,
   121→        name: f.name || user.name,
   122→        email: f.email || user.email,
   123→      }));
   124→    }
   125→  }, [user]);
   126→
   127→  const sub = subtotal();
   128→  const discount = couponState.status === "applied" ? couponState.discount ?? 0 : 0;
   129→  const afterDiscount = Math.max(0, sub - discount);
   130→  const shipping = afterDiscount === 0 || afterDiscount >= FREE_SHIPPING ? 0 : BASE_SHIPPING;
   131→  const total = afterDiscount + shipping;
   132→
   133→  async function applyCoupon() {
   134→    const code = couponInput.trim();
   135→    if (!code) return;
   136→    setCouponState({ status: "validating" });
   137→    try {
   138→      const res = await api.validateCoupon(code, sub);
   139→      if (res.valid) {
   140→        setCouponState({
   141→          status: "applied",
   142→          code: res.code,
   143→          description: res.description,
   144→          discount: res.discount,
   145→        });
   146→        toast.success(`Cupom ${res.code} aplicado! -${formatPrice(res.discount)}`);
   147→      } else {
   148→        setCouponState({
   149→          status: "error",
   150→          message: res.message ?? "Cupom não aplicável.",
   151→        });
   152→        toast.error(res.message ?? "Cupom não aplicável.");
   153→      }
   154→    } catch (err) {
   155→      const msg = err instanceof Error ? err.message : "Cupom inválido.";
   156→      setCouponState({ status: "error", message: msg });
   157→      toast.error(msg);
   158→    }
   159→  }
   160→
   161→  function removeCoupon() {
   162→    setCouponInput("");
   163→    setCouponState({ status: "idle" });
   164→  }
   165→
   166→  function update<K extends keyof FormState>(key: K, value: FormState[K]) {
   167→    setForm((f) => ({ ...f, [key]: value }));
   168→    if (errors[key]) {
   169→      setErrors((e) => {
   170→        const next = { ...e };
   171→        delete next[key];
   172→        return next;
   173→      });
   174→    }
   175→  }
   176→
   177→  function validate(): Errors {
   178→    const e: Errors = {};
   179→    if (!form.name.trim()) e.name = "Informe seu nome completo.";
   180→    if (!form.email.trim()) e.email = "Informe seu e-mail.";
   181→    else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email))
   182→      e.email = "E-mail inválido.";
   183→    if (!form.phone.trim() || form.phone.replace(/\D/g, "").length < 10)
   184→      e.phone = "Telefone inválido.";
   185→
   186→    if (!form.cep.trim() || form.cep.replace(/\D/g, "").length !== 8)
   187→      e.cep = "CEP inválido (8 dígitos).";
   188→    if (!form.street.trim()) e.street = "Informe a rua.";
   189→    if (!form.number.trim()) e.number = "Informe o número.";
   190→    if (!form.district.trim()) e.district = "Informe o bairro.";
   191→    if (!form.city.trim()) e.city = "Informe a cidade.";
   192→    if (!form.state.trim() || form.state.length !== 2)
   193→      e.state = "UF com 2 letras.";
   194→
   195→    if (form.payment === "card") {
   196→      if (form.cardNumber.replace(/\s/g, "").length < 13)
   197→        e.cardNumber = "Número do cartão inválido.";
   198→      if (!form.cardName.trim()) e.cardName = "Nome no cartão é obrigatório.";
   199→      if (!/^\d{2}\/\d{2}$/.test(form.cardExpiry))
   200→        e.cardExpiry = "Validade no formato MM/AA.";
   201→      if (form.cardCVV.length < 3) e.cardCVV = "CVV inválido.";
   202→    }
   203→    return e;
   204→  }
   205→
   206→  async function handleSubmit(ev: React.FormEvent) {
   207→    ev.preventDefault();
   208→    const found = validate();
   209→    setErrors(found);
   210→    if (Object.keys(found).length > 0) {
   211→      toast.error("Revise os campos destacados no formulário.");
   212→      return;
   213→    }
   214→
   215→    setSubmitting(true);
   216→    try {
   217→      const cardLast4 =
   218→        form.payment === "card"
   219→          ? form.cardNumber.replace(/\D/g, "").slice(-4)
   220→          : undefined;
   221→
   222→      const order = await api.createOrder({
   223→        items: items.map((i) => ({
   224→          productId: i.productId,
   225→          size: i.size,
   226→          quantity: i.quantity,
   227→        })),
   228→        customer: {
   229→          name: form.name.trim(),
   230→          email: form.email.trim(),
   231→          phone: form.phone.trim(),
   232→        },
   233→        address: {
   234→          cep: form.cep.trim(),
   235→          street: form.street.trim(),
   236→          number: form.number.trim(),
   237→          complement: form.complement.trim() || undefined,
   238→          district: form.district.trim(),
   239→          city: form.city.trim(),
   240→          state: form.state.trim().toUpperCase(),
   241→        },
   242→        payment: {
   243→          method: form.payment,
   244→          cardLast4,
   245→        },
   246→        couponCode:
   247→          couponState.status === "applied" ? couponState.code : undefined,
   248→      });
   249→
   250→      clear();
   251→      setLastOrder(order);
   252→      toast.success("Pedido confirmado! Preparando decolagem 🚀");
   253→      navigate("order-success");
   254→    } catch (err) {
   255→      const message =
   256→        err instanceof Error
   257→          ? err.message
   258→          : "Não foi possível finalizar a compra agora. Tente novamente.";
   259→      toast.error(message);
   260→    } finally {
   261→      setSubmitting(false);
   262→    }
   263→  }
   264→
   265→  // ---------- Empty cart ----------
   266→  if (items.length === 0) {
   267→    return (
   268→      <section className="mx-auto flex min-h-[70vh] max-w-3xl flex-col items-center justify-center px-4 py-16 text-center">
   269→        <motion.div
   270→          initial={{ opacity: 0, y: 20 }}
   271→          animate={{ opacity: 1, y: 0 }}
   272→          transition={{ duration: 0.5 }}
   273→          className="glass flex flex-col items-center gap-5 rounded-3xl p-10 sm:p-14"
   274→        >
   275→          <div className="relative flex h-24 w-24 items-center justify-center rounded-full bg-white/5">
   276→            <div className="absolute inset-0 rounded-full bg-[var(--neon-cyan)]/10 blur-2xl" />
   277→            <ShoppingBag className="h-11 w-11 text-[var(--neon-cyan)]" />
   278→          </div>
   279→          <div>
   280→            <h1 className="text-2xl font-bold sm:text-3xl">
   281→              Seu carrinho está vazio
   282→            </h1>
   283→            <p className="mx-auto mt-2 max-w-md text-sm text-muted-foreground">
   284→              Adicione um par de outra galáxia antes de seguir para o checkout.
   285→            </p>
   286→          </div>
   287→          <Button
   288→            onClick={() => navigate("products")}
   289→            className="rounded-full bg-gradient-to-r from-[var(--neon-cyan)] to-[var(--neon-violet)] px-6 py-3 text-sm font-bold text-black hover:opacity-90"
   290→          >
   291→            Explorar drops
   292→            <ArrowRight className="h-4 w-4" />
   293→          </Button>
   294→        </motion.div>
   295→      </section>
   296→    );
   297→  }
   298→
   299→  return (
   300→    <section className="mx-auto max-w-7xl px-4 py-10 sm:px-6 lg:py-14">
   301→      {/* Header */}
   302→      <motion.div
   303→        initial={{ opacity: 0, y: 16 }}
   304→        animate={{ opacity: 1, y: 0 }}
   305→        transition={{ duration: 0.5 }}
   306→        className="mb-8"
   307→      >
   308→        <button
   309→          onClick={() => navigate("products")}
   310→          className="mb-4 inline-flex items-center gap-1.5 text-sm text-muted-foreground transition hover:text-foreground"
   311→        >
   312→          <ArrowLeft className="h-4 w-4" />
   313→          Continuar comprando
   314→        </button>
   315→        <h1 className="text-3xl font-black tracking-tight sm:text-4xl">
   316→          <span className="text-gradient-neon">Checkout</span>
   317→        </h1>
   318→        <p className="mt-2 text-sm text-muted-foreground">
   319→          Pagamento seguro · Frete grátis acima de R$300
   320→        </p>
   321→      </motion.div>
   322→
   323→      <form
   324→        onSubmit={handleSubmit}
   325→        className="grid gap-8 lg:grid-cols-[1fr_380px]"
   326→        noValidate
   327→      >
   328→        {/* ---------- FORM (left) ---------- */}
   329→        <motion.div
   330→          initial={{ opacity: 0, y: 16 }}
   331→          animate={{ opacity: 1, y: 0 }}
   332→          transition={{ duration: 0.5, delay: 0.05 }}
   333→          className="space-y-6"
   334→        >
   335→          {/* Contato */}
   336→          <fieldset className="glass rounded-3xl p-6">
   337→            <legend className="flex items-center gap-2 px-2 text-sm font-bold uppercase tracking-wider text-[var(--neon-cyan)]">
   338→              <span className="flex h-6 w-6 items-center justify-center rounded-full bg-[var(--neon-cyan)]/15 text-xs font-black text-[var(--neon-cyan)]">
   339→                1
   340→              </span>
   341→              Contato
   342→            </legend>
   343→            <div className="mt-4 grid gap-4 sm:grid-cols-2">
   344→              <div className="sm:col-span-2">
   345→                <Label htmlFor="ck-name" className="mb-1.5 block text-sm">
   346→                  Nome completo
   347→                </Label>
   348→                <Input
   349→                  id="ck-name"
   350→                  value={form.name}
   351→                  onChange={(e) => update("name", e.target.value)}
   352→                  placeholder="Como devemos te chamar?"
   353→                  className={inputClass}
   354→                  autoComplete="name"
   355→                />
   356→                <FieldError message={errors.name} />
   357→              </div>
   358→              <div>
   359→                <Label htmlFor="ck-email" className="mb-1.5 block text-sm">
   360→                  E-mail
   361→                </Label>
   362→                <Input
   363→                  id="ck-email"
   364→                  type="email"
   365→                  value={form.email}
   366→                  onChange={(e) => update("email", e.target.value)}
   367→                  placeholder="voce@astrofeet.com"
   368→                  className={inputClass}
   369→                  autoComplete="email"
   370→                />
   371→                <FieldError message={errors.email} />
   372→              </div>
   373→              <div>
   374→                <Label htmlFor="ck-phone" className="mb-1.5 block text-sm">
   375→                  Telefone / WhatsApp
   376→                </Label>
   377→                <Input
   378→                  id="ck-phone"
   379→                  value={form.phone}
   380→                  onChange={(e) => update("phone", maskPhone(e.target.value))}
   381→                  placeholder="(11) 99999-9999"
   382→                  className={inputClass}
   383→                  inputMode="tel"
   384→                  autoComplete="tel"
   385→                />
   386→                <FieldError message={errors.phone} />
   387→              </div>
   388→            </div>
   389→          </fieldset>
   390→
   391→          {/* Entrega */}
   392→          <fieldset className="glass rounded-3xl p-6">
   393→            <legend className="flex items-center gap-2 px-2 text-sm font-bold uppercase tracking-wider text-[var(--neon-violet)]">
   394→              <span className="flex h-6 w-6 items-center justify-center rounded-full bg-[var(--neon-violet)]/15 text-xs font-black text-[var(--neon-violet)]">
   395→                2
   396→              </span>
   397→              Entrega
   398→            </legend>
   399→            <div className="mt-4 grid gap-4 sm:grid-cols-6">
   400→              <div className="sm:col-span-2">
   401→                <Label htmlFor="ck-cep" className="mb-1.5 block text-sm">
   402→                  CEP
   403→                </Label>
   404→                <Input
   405→                  id="ck-cep"
   406→                  value={form.cep}
   407→                  onChange={(e) => update("cep", maskCEP(e.target.value))}
   408→                  placeholder="00000-000"
   409→                  className={inputClass}
   410→                  inputMode="numeric"
   411→                  autoComplete="postal-code"
   412→                />
   413→                <FieldError message={errors.cep} />
   414→              </div>
   415→              <div className="sm:col-span-4">
   416→                <Label htmlFor="ck-street" className="mb-1.5 block text-sm">
   417→                  Rua
   418→                </Label>
   419→                <Input
   420→                  id="ck-street"
   421→                  value={form.street}
   422→                  onChange={(e) => update("street", e.target.value)}
   423→                  placeholder="Av. Via Láctea"
   424→                  className={inputClass}
   425→                  autoComplete="address-line1"
   426→                />
   427→                <FieldError message={errors.street} />
   428→              </div>
   429→              <div className="sm:col-span-2">
   430→                <Label htmlFor="ck-number" className="mb-1.5 block text-sm">
   431→                  Número
   432→                </Label>
   433→                <Input
   434→                  id="ck-number"
   435→                  value={form.number}
   436→                  onChange={(e) => update("number", e.target.value)}
   437→                  placeholder="42"
   438→                  className={inputClass}
   439→                  inputMode="numeric"
   440→                  autoComplete="address-line2"
   441→                />
   442→                <FieldError message={errors.number} />
   443→              </div>
   444→              <div className="sm:col-span-4">
   445→                <Label
   446→                  htmlFor="ck-complement"
   447→                  className="mb-1.5 block text-sm"
   448→                >
   449→                  Complemento{" "}
   450→                  <span className="text-muted-foreground">(opcional)</span>
   451→                </Label>
   452→                <Input
   453→                  id="ck-complement"
   454→                  value={form.complement}
   455→                  onChange={(e) => update("complement", e.target.value)}
   456→                  placeholder="Apto, bloco..."
   457→                  className={inputClass}
   458→                />
   459→              </div>
   460→              <div className="sm:col-span-3">
   461→                <Label htmlFor="ck-district" className="mb-1.5 block text-sm">
   462→                  Bairro
   463→                </Label>
   464→                <Input
   465→                  id="ck-district"
   466→                  value={form.district}
   467→                  onChange={(e) => update("district", e.target.value)}
   468→                  placeholder="Galáxia"
   469→                  className={inputClass}
   470→                />
   471→                <FieldError message={errors.district} />
   472→              </div>
   473→              <div className="sm:col-span-2">
   474→                <Label htmlFor="ck-city" className="mb-1.5 block text-sm">
   475→                  Cidade
   476→                </Label>
   477→                <Input
   478→                  id="ck-city"
   479→                  value={form.city}
   480→                  onChange={(e) => update("city", e.target.value)}
   481→                  placeholder="São Paulo"
   482→                  className={inputClass}
   483→                  autoComplete="address-level2"
   484→                />
   485→                <FieldError message={errors.city} />
   486→              </div>
   487→              <div className="sm:col-span-1">
   488→                <Label htmlFor="ck-state" className="mb-1.5 block text-sm">

   491→                <Input
   492→                  id="ck-state"
   493→                  value={form.state}
   494→                  onChange={(e) =>
   495→                    update(
   496→                      "state",
   497→                      e.target.value.replace(/[^a-zA-Z]/g, "").toUpperCase(),

   500→                  placeholder="SP"
   501→                  maxLength={2}
   502→                  className={inputClass}
   503→                  autoComplete="address-level1"
   504→                />
   505→                <FieldError message={errors.state} />

   508→          </fieldset>
   509→
   510→          {/* Pagamento */}
   511→          <fieldset className="glass rounded-3xl p-6">
   512→            <legend className="flex items-center gap-2 px-2 text-sm font-bold uppercase tracking-wider text-[var(--neon-magenta)]">
   513→              <span className="flex h-6 w-6 items-center justify-center rounded-full bg-[var(--neon-magenta)]/15 text-xs font-black text-[var(--neon-magenta)]">
   514→                3
   515→              </span>
   516→              Pagamento
   517→            </legend>
   518→
   519→            <RadioGroup
   520→              value={form.payment}
   521→              onValueChange={(v) => update("payment", v as PaymentMethod)}

   524→              <PaymentOption

   526→                title="Cartão de crédito"
   527→                hint="Em até 12x"
   528→                icon={<CreditCard className="h-5 w-5" />}
   529→                active={form.payment === "card"}
   530→              />
   531→              <PaymentOption
   532→                value="pix"
   533→                title="Pix"
   534→                hint="5% off à vista"
   535→                icon={<QrCode className="h-5 w-5" />}
   536→                active={form.payment === "pix"}
   537→              />
   538→              <PaymentOption
   539→                value="boleto"
   540→                title="Boleto"
   541→                hint="Vence em 3 dias"

   543→                active={form.payment === "boleto"}
   544→              />
   545→            </RadioGroup>
   546→
   547→            {/* Card fields */}
   548→            {form.payment === "card" && (
   549→              <motion.div
   550→                initial={{ opacity: 0, height: 0 }}
   551→                animate={{ opacity: 1, height: "auto" }}
   552→                transition={{ duration: 0.3 }}
   553→                className="mt-5 grid gap-4 sm:grid-cols-6"
   554→              >
   555→                <div className="sm:col-span-6">
   556→                  <Label
   557→                    htmlFor="ck-card-number"
   558→                    className="mb-1.5 block text-sm"
   559→                  >
   560→                    Número do cartão
   561→                  </Label>
   562→                  <Input
   563→                    id="ck-card-number"
   564→                    value={form.cardNumber}
   565→                    onChange={(e) =>
   566→                      update("cardNumber", maskCard(e.target.value))
   567→                    }
   568→                    placeholder="0000 0000 0000 0000"
   569→                    className={inputClass}
   570→                    inputMode="numeric"
   571→                    autoComplete="cc-number"
   572→                  />
   573→                  <FieldError message={errors.cardNumber} />
   574→                </div>
   575→                <div className="sm:col-span-6">
   576→                  <Label
   577→                    htmlFor="ck-card-name"

   579→                  >

   581→                  </Label>
   582→                  <Input
   583→                    id="ck-card-name"
   584→                    value={form.cardName}

   664→        {/* ---------- ORDER SUMMARY (right, sticky) ---------- */}

   717→            <Separator className="my-4 bg-white/10" />
   718→
   719→            {/* Coupon */}

   721→              {couponState.status === "applied" ? (
   722→                <div className="flex items-center justify-between gap-2 rounded-xl border border-emerald-500/30 bg-emerald-500/10 px-3 py-2.5">

   748→                  </div>
   749→                </div>
   750→              ) : (

   752→                  <div className="relative flex-1">
   753→                    <Ticket className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   754→                    <input
   755→                      type="text"
   756→                      value={couponInput}
   757→                      onChange={(e) => {
   758→                        setCouponInput(e.target.value.toUpperCase());
   759→                        if (couponState.status === "error")
   760→                          setCouponState({ status: "idle" });
   761→                      }}
   762→                      placeholder="Cupom de desconto"
   763→                      className="h-10 w-full rounded-xl border border-white/10 bg-white/5 pl-9 pr-3 text-sm uppercase tracking-wider outline-none placeholder:text-muted-foreground/60 focus:border-[var(--neon-cyan)]"
   764→                    />

   766→                  <button

   770→                      !couponInput.trim() || couponState.status === "validating"
   771→                    }

   781→              )}
   782→              {couponState.status === "error" && (
   783→                <p className="mt-1.5 text-xs text-rose-400">
   784→                  {couponState.message}

   804→              <div className="flex justify-between text-muted-foreground">

   808→                </span>
   809→              </div>

   816→              <div className="flex justify-between text-muted-foreground">

   859→                </>

   868→            {/* Trust row */}

   878→            </div>
   879→          </div>
