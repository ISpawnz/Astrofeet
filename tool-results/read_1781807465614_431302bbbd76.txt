   107→function paymentMethodInfo(method: string): {
   108→  label: string;
   109→  icon: React.ReactNode;
   110→} {
   111→  switch (method) {
   112→    case "card":
   113→      return { label: "Cartão de crédito", icon: <CreditCard className="h-4 w-4" /> };
   114→    case "pix":
   115→      return { label: "Pix", icon: <QrCode className="h-4 w-4" /> };
   116→    case "boleto":
   117→      return { label: "Boleto", icon: <Barcode className="h-4 w-4" /> };
   118→    default:
   119→      return { label: method || "Pagamento", icon: <CreditCard className="h-4 w-4" /> };
   120→  }
   121→}

   123→function getInitial(name: string | undefined): string {
   124→  if (!name) return "✦";
   125→  const trimmed = name.trim();
   126→  if (!trimmed) return "✦";
   127→  return trimmed[0]!.toUpperCase();
   128→}
   129→
   130→function itemsSummary(order: Order): string {
   131→  const items = order.items;
   132→  if (items.length === 0) return "Sem itens";
   133→  const firstTwo = items.slice(0, 2).map((i) => `${i.name} (Tam ${i.size})`);
   134→  const more = items.length - 2;
   135→  const base = firstTwo.join(", ");
   136→  if (more > 0) return `${base} +${more} mais`;
   137→  return base;
   138→}
   139→
   140→function itemCount(order: Order): number {
   141→  return order.items.reduce((n, i) => n + i.quantity, 0);
   142→}
   143→
   144→function isCancelled(status: string): boolean {
   145→  return status === "cancelled";

   776→function ProfileTab({ user }: { user: PublicUser }) {

   779→  const roleLabel = user.role === "admin" ? "Comando" : "Explorador";
   780→  const roleAccent =
   781→    user.role === "admin" ? "var(--neon-magenta)" : "var(--neon-cyan)";
   782→
   783→  // Editable name state
   784→  const [editingName, setEditingName] = useState(false);
   785→  const [nameValue, setNameValue] = useState(user.name);
   786→  const [savingName, setSavingName] = useState(false);
   787→
   788→  // Password change state
   789→  const [currentPassword, setCurrentPassword] = useState("");
   790→  const [newPassword, setNewPassword] = useState("");
   791→  const [confirmPassword, setConfirmPassword] = useState("");
   792→  const [savingPassword, setSavingPassword] = useState(false);
   793→  const [showCurrent, setShowCurrent] = useState(false);
   794→  const [showNew, setShowNew] = useState(false);
   795→  const [showConfirm, setShowConfirm] = useState(false);
   796→
   797→  useEffect(() => {
   798→    setNameValue(user.name);
   799→  }, [user.name]);
   800→
   801→  async function saveName() {
   802→    const trimmed = nameValue.trim();
   803→    if (trimmed.length < 2) {
   804→      toast.error("Nome precisa ter ao menos 2 caracteres.");
   805→      return;
   806→    }
   807→    if (trimmed === user.name) {
   808→      setEditingName(false);
   809→      return;
   810→    }
   811→    setSavingName(true);
   812→    try {
   813→      const updated = await api.updateProfile(trimmed);
   814→      setUser(updated);
   815→      toast.success("Nome atualizado com sucesso!");
   816→      setEditingName(false);
   817→    } catch (err) {
   818→      toast.error(
   819→        err instanceof Error ? err.message : "Não foi possível salvar.",
   820→      );
   821→    } finally {
   822→      setSavingName(false);
   823→    }
   824→  }

   826→  function resetPasswordFields() {

   828→    setNewPassword("");
   829→    setConfirmPassword("");

   834→      toast.error("Informe sua senha atual.");

   838→      toast.error("A nova senha precisa ter ao menos 6 caracteres.");
   839→      return;
   840→    }
   841→    if (newPassword === currentPassword) {
   842→      toast.error("A nova senha precisa ser diferente da atual.");
   843→      return;
   844→    }
   845→    if (newPassword !== confirmPassword) {
   846→      toast.error("As senhas não coincidem.");
   847→      return;
   848→    }
   849→    setSavingPassword(true);
   850→    try {
   851→      await api.changePassword(currentPassword, newPassword);

   853→      resetPasswordFields();
   854→    } catch (err) {
   855→      toast.error(
   856→        err instanceof Error
   857→          ? err.message
   858→          : "Não foi possível atualizar a senha agora.",
   859→      );
   860→    } finally {
   861→      setSavingPassword(false);
   862→    }
   863→  }
   864→
   865→  return (
