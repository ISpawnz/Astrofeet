   107→function paymentMethodInfo(method: string): {
   108→  label: string;
   109→  icon: React.ReactNode;
   110→} {
   111→  switch (method) {
   112→    case "card":
   113→      return { label: "Cartão de crédito", icon: <CreditCard className="h-4 w-4" /> };
   114→    case "pix":
   115→      return { label: "Pix", icon: <QrCode className="h-4 w-4" /> };
   116→    case "boleto":
   117→      return { label: "Boleto", icon: <Barcode className="h-4 w-4" /> };
   118→    default:
   119→      return { label: method || "Pagamento", icon: <CreditCard className="h-4 w-4" /> };
   120→  }
   121→}

   123→function getInitial(name: string | undefined): string {
   124→  if (!name) return "✦";
   125→  const trimmed = name.trim();
   126→  if (!trimmed) return "✦";
   127→  return trimmed[0]!.toUpperCase();
   128→}
   129→
   130→function itemsSummary(order: Order): string {
   131→  const items = order.items;
   132→  if (items.length === 0) return "Sem itens";
   133→  const firstTwo = items.slice(0, 2).map((i) => `${i.name} (Tam ${i.size})`);
   134→  const more = items.length - 2;
   135→  const base = firstTwo.join(", ");
   136→  if (more > 0) return `${base} +${more} mais`;
   137→  return base;
   138→}
   139→
   140→function itemCount(order: Order): number {
   141→  return order.items.reduce((n, i) => n + i.quantity, 0);
   142→}

   144→function isCancelled(status: string): boolean {
   145→  return status === "cancelled";

   899→function ProfileTab({ user }: { user: PublicUser }) {

   902→  const roleLabel = user.role === "admin" ? "Comando" : "Explorador";
   903→  const roleAccent =
   904→    user.role === "admin" ? "var(--neon-magenta)" : "var(--neon-cyan)";
   905→
   906→  // Editable name state
   907→  const [editingName, setEditingName] = useState(false);
   908→  const [nameValue, setNameValue] = useState(user.name);
   909→  const [savingName, setSavingName] = useState(false);
   910→
   911→  // Password change state
   912→  const [currentPassword, setCurrentPassword] = useState("");
   913→  const [newPassword, setNewPassword] = useState("");
   914→  const [confirmPassword, setConfirmPassword] = useState("");

   918→  const [showConfirm, setShowConfirm] = useState(false);
   919→
   920→  useEffect(() => {
   921→    setNameValue(user.name);

   924→  async function saveName() {
   925→    const trimmed = nameValue.trim();
   926→    if (trimmed.length < 2) {
   927→      toast.error("Nome precisa ter ao menos 2 caracteres.");
   928→      return;

   931→      setEditingName(false);

   936→      const updated = await api.updateProfile(trimmed);

   977→    } catch (err) {

   979→        err instanceof Error
   980→          ? err.message
   981→          : "Não foi possível atualizar a senha agora.",
   982→      );
   983→    } finally {
   984→      setSavingPassword(false);
   985→    }
